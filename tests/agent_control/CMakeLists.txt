cmake_minimum_required(VERSION 3.16)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(orca_agent_control_tests LANGUAGES CXX)
endif ()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost 1.75 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
if (NOT TARGET Catch2::Catch2WithMain)
    add_subdirectory(../catch2 "${CMAKE_CURRENT_BINARY_DIR}/catch2")
endif ()
get_filename_component(ORCA_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(agent_transport_tests test_transport.cpp
    "${ORCA_ROOT}/src/slic3r/AgentControl/RpcServer.cpp")
target_include_directories(agent_transport_tests PRIVATE "${ORCA_ROOT}/src")
target_include_directories(agent_transport_tests SYSTEM PRIVATE "${ORCA_ROOT}/deps_src" ${Boost_INCLUDE_DIRS})
target_compile_definitions(agent_transport_tests PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(agent_transport_tests PRIVATE OpenSSL::Crypto Threads::Threads Catch2::Catch2WithMain)
if(WIN32)
    target_link_libraries(agent_transport_tests PRIVATE advapi32 ws2_32 mswsock)
endif()
add_executable(agent_commands_tests test_commands.cpp
    "${ORCA_ROOT}/src/slic3r/AgentControl/Commands.cpp")
target_include_directories(agent_commands_tests PRIVATE "${ORCA_ROOT}/src")
target_include_directories(agent_commands_tests SYSTEM PRIVATE "${ORCA_ROOT}/deps_src")
target_link_libraries(agent_commands_tests PRIVATE Threads::Threads Catch2::Catch2WithMain)
enable_testing()
add_test(NAME agent_commands COMMAND agent_commands_tests)
add_test(NAME agent_transport COMMAND agent_transport_tests)
set_tests_properties(agent_transport PROPERTIES TIMEOUT 20)
set_tests_properties(agent_commands agent_transport PROPERTIES LABELS "AgentControl")

add_executable(agent_print_journal_tests test_print_journal.cpp
    "${ORCA_ROOT}/src/slic3r/AgentControl/PrintJournal.cpp")
target_include_directories(agent_print_journal_tests PRIVATE "${ORCA_ROOT}/src")
target_include_directories(agent_print_journal_tests SYSTEM PRIVATE "${ORCA_ROOT}/deps_src")
target_link_libraries(agent_print_journal_tests PRIVATE Threads::Threads Catch2::Catch2WithMain)
if(WIN32)
    target_link_libraries(agent_print_journal_tests PRIVATE advapi32)
endif()
add_test(NAME agent_print_journal COMMAND agent_print_journal_tests)
set_tests_properties(agent_print_journal PROPERTIES LABELS "AgentControl" TIMEOUT 20)

add_executable(agent_file_access_tests test_file_access.cpp
    "${ORCA_ROOT}/src/slic3r/AgentControl/FileAccess.cpp")
target_include_directories(agent_file_access_tests PRIVATE "${ORCA_ROOT}/src")
target_include_directories(agent_file_access_tests SYSTEM PRIVATE "${ORCA_ROOT}/deps_src")
target_link_libraries(agent_file_access_tests PRIVATE OpenSSL::Crypto Threads::Threads Catch2::Catch2WithMain)
if(WIN32)
    target_link_libraries(agent_file_access_tests PRIVATE advapi32)
endif()
add_test(NAME agent_file_access COMMAND agent_file_access_tests)
set_tests_properties(agent_file_access PROPERTIES LABELS "AgentControl" TIMEOUT 30)

add_executable(agent_print_materials_tests test_print_materials.cpp)
target_include_directories(agent_print_materials_tests PRIVATE "${ORCA_ROOT}/src")
target_include_directories(agent_print_materials_tests SYSTEM PRIVATE "${ORCA_ROOT}/deps_src")
target_link_libraries(agent_print_materials_tests PRIVATE Catch2::Catch2WithMain)
add_test(NAME agent_print_materials COMMAND agent_print_materials_tests)
set_tests_properties(agent_print_materials PROPERTIES LABELS "AgentControl" TIMEOUT 20)
